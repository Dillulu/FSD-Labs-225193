/* ============================================================================
   FSD LAB 04 - Core Logic (Hamza Tahir - 225193)
   ============================================================================ */
document.addEventListener('DOMContentLoaded', () => {
  // 1. Copy/Cut Protection
  ['copy', 'cut'].forEach(ev => {
    productDescription.addEventListener(ev, (e) => {
      e.preventDefault();
      showProtectionNotice(`Product description text cannot be ${ev}ed!`);
    });
  });

  // 2. Real-Time Engraving with character feedback
  engravingInput.addEventListener('input', function () {
    let cleanVal = this.value.replace(/[^a-zA-Z0-9\s\-_.&!❤️★]/g, '');
    this.value = cleanVal;
    engravingTextDisplay.textContent = cleanVal;
    charCount.textContent = `${cleanVal.length}/18`;
  });

  // 3. 3D Perspective Tilt on Pointer Move
  productPreview.addEventListener('mousemove', (e) => {
    const rect = productPreview.getBoundingClientRect();
    const x = e.clientX - rect.left, y = e.clientY - rect.top;
    const tiltX = ((y - rect.height/2) / (rect.height/2)) * -12;
    const tiltY = ((x - rect.width/2) / (rect.width/2)) * 12;
    productPreview.style.transform = `perspective(700px) rotateX(${tiltX.toFixed(2)}deg) rotateY(${tiltY.toFixed(2)}deg) scale(1.03)`;
    previewOverlay.style.background = `radial-gradient(circle at ${(x/rect.width)*100}% ${(y/rect.height)*100}%, rgba(255,255,255,0.35) 0%, transparent 70%)`;
  });

  // 4. HTML5 Drag & Drop Implementation
  productPreview.addEventListener('dragstart', (e) => {
    productPreview.classList.add('dragging');
    e.dataTransfer.setData('text/plain', 'custom-mug');
  });
  dropZone.addEventListener('dragover', (e) => { e.preventDefault(); dropZone.classList.add('over'); });
  dropZone.addEventListener('dragleave', () => dropZone.classList.remove('over'));
  dropZone.addEventListener('drop', (e) => {
    e.preventDefault();
    dropZone.classList.remove('over');
    addItemToCart();
  });

  // 10 & 11. Simulated Payment Gateway & Strict Regex Validation
  paymentForm.addEventListener('submit', (e) => {
    e.preventDefault();
    if (!validateCardholderName() || !validateCardNumber() || !validateExpiryDate() || !validateCVV()) return;
    paymentForm.hidden = true; processingMsg.hidden = false;
    setTimeout(() => {
      processingMsg.hidden = true; successMsg.hidden = false;
      activatePremiumStatus();
      setTimeout(closePaymentModalWindow, 1600);
    }, 1500);
  });
});
